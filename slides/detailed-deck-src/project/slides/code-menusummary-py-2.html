def leaves(mods, name, seen=None):
    if seen is None: seen = set()
    out = []
    typ, body = mods.get(name, ("?", ""))
    for mem in members(body):
        if mem in seen: continue
        seen.add(mem)
        t2 = mods.get(mem, ("?", ""))[0]
        if t2 in ("Sequence", "Path"):
            out += leaves(mods, mem, seen)
        elif t2 == "EDFilter":
            out.append(mem)
    return out
